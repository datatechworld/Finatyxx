model SupplierBill {
  id                    String                      @id @default(uuid()) @db.Uuid
  tenantId              String                      @db.Uuid
  branchId              String                      @db.Uuid
  supplierId            String                      @db.Uuid
  purchaseOrderId       String?                     @db.Uuid
  billNumber            String                      @db.VarChar(80)
  supplierInvoiceNumber String?                     @db.VarChar(120)
  billDate              DateTime                    @db.Date
  dueDate               DateTime                    @db.Date
  currencyCode          String                      @db.Char(3)
  exchangeRate          Decimal                     @default(1) @db.Decimal(20, 8)
  subtotal              Decimal                     @db.Decimal(20, 4)
  taxTotal              Decimal                     @default(0) @db.Decimal(20, 4)
  total                 Decimal                     @db.Decimal(20, 4)
  amountPaid            Decimal                     @default(0) @db.Decimal(20, 4)
  balanceDue            Decimal                     @db.Decimal(20, 4)
  status                DocumentStatus              @default(DRAFT)
  postedAt              DateTime?                   @db.Timestamptz(6)
  createdAt             DateTime                    @default(now()) @db.Timestamptz(6)
  supplier              Supplier                    @relation(fields: [tenantId, supplierId], references: [tenantId, id], onDelete: Restrict)
  purchaseOrder         PurchaseOrder?              @relation(fields: [tenantId, purchaseOrderId], references: [tenantId, id], onDelete: Restrict)
  items                 SupplierBillItem[]
  allocations           SupplierPaymentAllocation[]

  @@unique([tenantId, billNumber])
  @@unique([tenantId, id])
  @@index([tenantId, supplierId, dueDate])
  @@map("supplier_bills")
}
